function page(id,button){

    document.querySelectorAll(".page").forEach(function(p){
        p.classList.remove("active");
    });

    const target=document.getElementById(id);

    if(target){
        target.classList.add("active");
    }

    if(button){

        document.querySelectorAll(".nav").forEach(function(n){
            n.classList.remove("active");
        });

        button.classList.add("active");
    }
}

let toastTimer;

function showToast(message){

    const toast=document.getElementById("toast");

    toast.textContent=message;
    toast.style.display="block";

    clearTimeout(toastTimer);

    toastTimer=setTimeout(function(){

        toast.style.display="none";

    },1800);
}

/* =========================================================

function copyText(id){

    const element=document.getElementById(id);

    navigator.clipboard.writeText(element.value)
    .then(function(){

        showToast("📋 Copied!");

    })
    .catch(function(){

        element.select();
        document.execCommand("copy");

        showToast("📋 Copied!");

    });

}

/* =========================================================
   GIVE GENERATOR
========================================================= */

function generateGive(){

    const version=getVersion();

    const item=
        cleanItem(
            document.getElementById("giveItem").value
        ) || "diamond";

    const amount=
        Math.max(
            1,
            Number(
                document.getElementById("giveAmount").value
            ) || 1
        );

    const name=
        document.getElementById("giveName").value.trim();

    const enchantments=
        parseEnchantments(
            document.getElementById("giveEnchant").value
        );


    let command="";


    if(version==="1.21"){

        let components=[];

        if(name){

            components.push(
                "minecraft:custom_name='"+
                jsonText(name)+
                "'"
            );

        }

        if(enchantments.length){

            let levels=[];

            enchantments.forEach(function(e){

                levels.push(
                    "minecraft:"+
                    e.id+
                    ":"+
                    e.level
                );

            });

            components.push(
                "minecraft:enchantments={levels:{"+
                levels.join(",")+
                "}}"
            );

        }

        command=
            "/give @p minecraft:"+
            item;

        if(components.length){

            command+=
                "["+
                components.join(",")+
                "]";

        }

        command+=" "+amount;

    }

    else{

        command=
            "/give @p minecraft:"+
            item+
            " "+
            amount;

        let nbt=[];

        if(name){

            nbt.push(
                "display:{Name:"+
                jsonText(name)+
                "}"
            );

        }

        if(enchantments.length){

            let list=[];

            enchantments.forEach(function(e){

                list.push(
                    "{id:\"minecraft:"+
                    e.id+
                    "\",lvl:"+
                    e.level+
                    "s}"
                );

            });

            nbt.push(
                "ench:["+
                list.join(",")+
                "]"
            );

        }

        if(nbt.length){

            command+=" {"+nbt.join(",")+"}";

        }

    }


    document.getElementById("giveResult").value=command;

    document.getElementById(
        "giveOutput"
    ).style.display="block";

}

/* =========================================================
   ADD TRADE
========================================================= */

function addTrade(data){

    const id=tradeID++;


    const trade=data || {

        cost1:"emerald",
        cost1Amount:1,

        cost2:"",
        cost2Amount:1,

        reward:"diamond",
        rewardAmount:1,

        maxUses:999999,
        xp:10,

        priceMultiplier:0.05,

        disabled:false

    };


    trades.push({
        id:id,
        ...trade
    });


    renderTrades();

}


/* =========================================================
   REMOVE TRADE
========================================================= */

function removeTrade(id){

    trades=
        trades.filter(function(t){

            return t.id!==id;

        });


    renderTrades();

}


/* =========================================================
   MOVE TRADE
========================================================= */

function moveTrade(id,direction){

    const index=
        trades.findIndex(function(t){

            return t.id===id;

        });


    if(index<0)return;


    const newIndex=
        index+direction;


    if(
        newIndex<0 ||
        newIndex>=trades.length
    ){

        return;

    }


    const temp=trades[index];

    trades[index]=trades[newIndex];

    trades[newIndex]=temp;


    renderTrades();

}


/* =========================================================
   RESET TRADES
========================================================= */

function resetTrades(){

    if(!confirm("Reset all trades?")){

        return;

    }


    trades=[];
    tradeID=0;

    addTrade({

        cost1:"emerald",
        cost1Amount:1,

        cost2:"",
        cost2Amount:1,

        reward:"diamond",
        rewardAmount:1,

        maxUses:999999,
        xp:10,

        priceMultiplier:0.05,

        disabled:false

    });


    showToast("Trades reset");

}

function updateTrade(id,key,value){

    const trade=
        trades.find(function(t){

            return t.id===id;

        });


    if(!trade)return;

    trade[key]=value;

}

/* =========================================================
   OLD ITEM STACK
   Used by older Minecraft versions.
========================================================= */

function oldItemStack(item,amount){

    return (
        "{id:\"minecraft:"+
        cleanItem(item)+
        "\",Count:"+
        Math.max(1,Number(amount)||1)+
        "b}"
    );

}


/* =========================================================
   MODERN ITEM STACK
   Used by 1.20.5+ / 1.21 style.
========================================================= */

function modernItemStack(item,amount){

    return (
        "{id:\"minecraft:"+
        cleanItem(item)+
        "\",count:"+
        Math.max(1,Number(amount)||1)+
        "}"
    );

}

/* =========================================================
   GENERATE OLD TRADE
========================================================= */

function generateOldTrade(t){

    let result=
        "{buy:"+
        oldItemStack(
            t.cost1,
            t.cost1Amount
        );


    if(t.cost2){

        result+=
            ",buyB:"+
            oldItemStack(
                t.cost2,
                t.cost2Amount
            );

    }


    result+=
        ",sell:"+
        oldItemStack(
            t.reward,
            t.rewardAmount
        );


    result+=
        ",maxUses:"+
        Math.max(
            1,
            Number(t.maxUses)||999999
        );


    result+=
        ",rewardExp:"+
        (Number(t.xp)>0?"true":"false");


    result+="}";


    return result;

}

/* =========================================================
   HELPERS
========================================================= */

function cleanItem(value){

    return String(value||"")
        .trim()
        .toLowerCase()
        .replace(/^minecraft:/,"")
        .replace(/\s+/g,"_");

}


function safeAttr(value){

    return String(value??"")
        .replace(/&/g,"&amp;")
        .replace(/"/g,"&quot;")
        .replace(/</g,"&lt;")
        .replace(/>/g,"&gt;");

}


function jsonText(text){

    return (
        "'"+
        JSON.stringify({
            text:String(text)
        })+
        "'"
    );

}


function parseEnchantments(value){

    const result=[];

    if(!value)return result;


    String(value)
        .split(",")
        .forEach(function(part){

            const x=
                part.trim().split(":");


            if(x.length<2)return;


            const id=
                cleanItem(x[0]);
